cmake_minimum_required(VERSION 3.20)
project(BetterCaptionsCore VERSION 0.1.0 LANGUAGES CXX)

# This deliberately builds no Premiere plug-in. An SDK adapter requires Adobe's
# licensed headers, sample resource/build steps and actual host acceptance.
option(BC_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(CTest)

add_library(better_captions_core STATIC core/caption_core.cpp)
target_include_directories(better_captions_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/core")
target_compile_features(better_captions_core PUBLIC cxx_std_17)
set_target_properties(better_captions_core PROPERTIES CXX_EXTENSIONS OFF)

if(MSVC)
  target_compile_options(better_captions_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(better_captions_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BC_ENABLE_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT MSVC)
    target_compile_options(better_captions_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(better_captions_core PUBLIC -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "BC_ENABLE_SANITIZERS requires a GCC/Clang compiler with address and undefined sanitizers")
  endif()
endif()

if(BUILD_TESTING)
  add_executable(caption_core_tests tests/caption_core_test.cpp)
  target_link_libraries(caption_core_tests PRIVATE better_captions_core)
  add_test(NAME caption_core_tests COMMAND caption_core_tests)
  add_executable(line_position_tests tests/line_position_test.cpp)
  target_link_libraries(line_position_tests PRIVATE better_captions_core)
  add_test(NAME line_position_tests COMMAND line_position_tests)
endif()

if(APPLE)
  enable_language(OBJCXX)
  add_library(better_captions_macos STATIC macos/caption_rasterizer.mm)
  target_compile_features(better_captions_macos PUBLIC cxx_std_17)
  set_target_properties(better_captions_macos PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS OFF)
  target_include_directories(better_captions_macos PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/macos")
  target_link_libraries(better_captions_macos PUBLIC better_captions_core "-framework CoreText" "-framework CoreGraphics" "-framework CoreFoundation")
  if(BUILD_TESTING)
    add_executable(caption_rasterizer_test macos/rasterizer_test.mm)
    set_target_properties(caption_rasterizer_test PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
    target_link_libraries(caption_rasterizer_test PRIVATE better_captions_macos)
    add_test(NAME caption_rasterizer_test COMMAND caption_rasterizer_test)
    add_executable(caption_scroll_spacing_test macos/scroll_spacing_test.mm)
    set_target_properties(caption_scroll_spacing_test PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
    target_link_libraries(caption_scroll_spacing_test PRIVATE better_captions_macos)
    add_test(NAME caption_scroll_spacing_test COMMAND caption_scroll_spacing_test)
  endif()
endif()
